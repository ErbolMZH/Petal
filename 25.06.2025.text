SELECT 
    n.id AS notification_id,
    -- Вытаскиваем ID из текста для наглядности в отчете
    REGEXP_SUBSTR(n.content_summary, 'ID\s+субъекта:\s*([a-zA-Z0-9]+)', 1, 1, NULL, 1) AS parsed_subject_id,
    -- Вытаскиваем дату из текста уведомления
    TO_DATE(REGEXP_SUBSTR(n.content_summary, 'Дата:\s*([0-9]{4}-[0-9]{2}-[0-9]{2}\s+[0-9]{2}:[0-9]{2}:[0-9]{2})', 1, 1, NULL, 1), 'YYYY-MM-DD HH24:MI:SS') AS notification_date,
    s.id AS subject_id,
    s.updated_date AS subject_updated_date
FROM 
    notifications n
JOIN 
    subject s 
    -- Вытаскиваем ID из текста уведомления и связываем с нормальным id из таблицы subject
    ON s.id = REGEXP_SUBSTR(n.content_summary, 'ID\s+субъекта:\s*([a-zA-Z0-9]+)', 1, 1, NULL, 1)
WHERE 
    n.n_type = 'subject_categories_update'
    -- Условие: в уведомлении дата свежее, чем в самой таблице subject (с учетом сдвига на 6 часов)
    AND TO_DATE(REGEXP_SUBSTR(n.content_summary, 'Дата:\s*([0-9]{4}-[0-9]{2}-[0-9]{2}\s+[0-9]{2}:[0-9]{2}:[0-9]{2})', 1, 1, NULL, 1), 'YYYY-MM-DD HH24:MI:SS') 
        > (s.updated_date + INTERVAL '6' HOUR);



        SELECT 
    n.id AS notification_id,
    -- Вытаскиваем ID (ищем всё, что идет после 'ID субъекта:' до ближайшего пробела или конца строки)
    substring(n.content_summary FROM 'ID\s+субъекта:\s*([^\s]+)') AS parsed_subject_id,
    -- Вытаскиваем дату и сразу кастуем её в TIMESTAMP
    substring(n.content_summary FROM 'Дата:\s*([0-9]{4}-[0-9]{2}-[0-9]{2}\s+[0-9]{2}:[0-9]{2}:[0-9]{2})')::timestamp AS notification_date,
    s.id AS subject_id,
    s.updated_date AS subject_updated_date
FROM 
    notifications n
JOIN 
    subject s 
    -- Приводим обе стороны к text (на случай, если s.id имеет тип uuid или int, а из текста мы получили строку)
    ON s.id::text = substring(n.content_summary FROM 'ID\s+субъекта:\s*([^\s]+)')::text
WHERE 
    n.n_type = 'subject_categories_update'
    -- Сравниваем даты: к отстающей дате subject прибавляем 6 часов
    AND substring(n.content_summary FROM 'Дата:\s*([0-9]{4}-[0-9]{2}-[0-9]{2}\s+[0-9]{2}:[0-9]{2}:[0-9]{2})')::timestamp 
        > (s.updated_date + INTERVAL '6 hours');



SELECT 
    n.id AS notification_id,
    n.n_type,
    -- Вытаскиваем ID и дату для контроля в результате
    substring(n.content_summary FROM 'ID\s+субъекта:\s*([^\s]+)') AS parsed_subject_id,
    substring(n.content_summary FROM 'Дата:\s*([0-9]{4}-[0-9]{2}-[0-9]{2}\s+[0-9]{2}:[0-9]{2}:[0-9]{2})')::timestamp AS notification_date,
    s.id AS subject_id,
    s.created_date,
    s.updated_date,
    s.deleted_date
FROM 
    notifications n
JOIN 
    subject s 
    ON s.id::text = substring(n.content_summary FROM 'ID\s+субъекта:\s*([^\s]+)')::text
WHERE 
    -- 1. Фильтруем только нужные типы уведомлений
    n.n_type IN ('subject_categories_update', 'subject_categories_delete', 'subject_create', 'subject_delete', 'subject_update')
    
    -- 2. Динамически проверяем нужную колонку в зависимости от n_type
    AND substring(n.content_summary FROM 'Дата:\s*([0-9]{4}-[0-9]{2}-[0-9]{2}\s+[0-9]{2}:[0-9]{2}:[0-9]{2})')::timestamp 
        != CASE 
            -- Для создания смотрим на created_date
            WHEN n.n_type = 'subject_create' 
                THEN s.created_date + INTERVAL '6 hours'
            
            -- Для удалений смотрим на deleted_date
            WHEN n.n_type IN ('subject_delete', 'subject_categories_delete') 
                THEN s.deleted_date + INTERVAL '6 hours'
            
            -- Для всех видов обновлений смотрим на updated_date
            WHEN n.n_type IN ('subject_update', 'subject_categories_update') 
                THEN s.updated_date + INTERVAL '6 hours'
           END;




SELECT 
    n.id AS notification_id,
    n.n_type,
    substring(n.content_summary FROM 'ID\s+субъекта:\s*([^\s]+)') AS parsed_subject_id,
    substring(n.content_summary FROM 'Дата:\s*([0-9]{4}-[0-9]{2}-[0-9]{2}\s+[0-9]{2}:[0-9]{2}:[0-9]{2})')::timestamp AS notification_date,
    s.id AS subject_id,
    CASE 
        WHEN n.n_type = 'subject_create' THEN s.created_date
        WHEN n.n_type IN ('subject_delete', 'subject_categories_delete') THEN s.deleted_date
        ELSE s.updated_date
    END AS db_date,
    -- Высчитываем чистую разницу в часах для наглядности в отчете
    ABS(EXTRACT(EPOCH FROM (
        substring(n.content_summary FROM 'Дата:\s*([0-9]{4}-[0-9]{2}-[0-9]{2}\s+[0-9]{2}:[0-9]{2}:[0-9]{2})')::timestamp 
        - 
        CASE 
            WHEN n.n_type = 'subject_create' THEN s.created_date
            WHEN n.n_type IN ('subject_delete', 'subject_categories_delete') THEN s.deleted_date
            ELSE s.updated_date
        END
    ))) / 3600 AS hours_difference
FROM 
    notifications n
JOIN 
    subject s 
    ON s.id::text = substring(n.content_summary FROM 'ID\s+субъекта:\s*([^\s]+)')::text
WHERE 
    n.n_type IN ('subject_categories_update', 'subject_categories_delete', 'subject_create', 'subject_delete', 'subject_update')
    
    -- Условие рассинхронизации:
    -- Если разница между уведомлением и базой НЕ входит в диапазон от 4 до 7 часов,
    -- значит это реальная просрочка обновления.
    AND ABS(EXTRACT(EPOCH FROM (
        substring(n.content_summary FROM 'Дата:\s*([0-9]{4}-[0-9]{2}-[0-9]{2}\s+[0-9]{2}:[0-9]{2}:[0-9]{2})')::timestamp 
        - 
        CASE 
            WHEN n.n_type = 'subject_create' THEN s.created_date
            WHEN n.n_type IN ('subject_delete', 'subject_categories_delete') THEN s.deleted_date
            ELSE s.updated_date
        END
    ))) / 3600 NOT BETWEEN 4.0 AND 7.0;
