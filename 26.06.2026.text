WITH parsed_notifications AS (
    SELECT 
        n.id AS notification_id,
        n.n_type,
        -- Вытаскиваем данные из текста уведомления
        substring(n.content_summary FROM 'ID\s+субъекта:\s*([^\s]+)') AS parsed_subject_id,
        substring(n.content_summary FROM 'Дата:\s*([0-9]{4}-[0-9]{2}-[0-9]{2}\s+[0-9]{2}:[0-9]{2}:[0-9]{2})')::timestamp AS notification_date,
        
        -- Выводим ID и даты из таблицы subject
        s.id AS subject_id,
        s.created_date AS db_created_date,
        s.updated_date AS db_updated_date,
        s.deleted_date AS db_deleted_date,
        
        -- Определяем, с какой именно датой из БД мы сравниваем в зависимости от типа
        CASE 
            WHEN n.n_type = 'subject_create' THEN s.created_date
            WHEN n.n_type IN ('subject_delete', 'subject_categories_delete') THEN s.deleted_date
            ELSE s.updated_date
        END AS target_db_date
    FROM 
        notifications n
    JOIN 
        subject s 
        ON s.id::text = substring(n.content_summary FROM 'ID\s+субъекта:\s*([^\s]+)')::text
    WHERE 
        n.n_type IN ('subject_categories_update', 'subject_categories_delete', 'subject_create', 'subject_delete', 'subject_update')
)
SELECT 
    notification_id,
    n_type,
    parsed_subject_id,
    notification_date,
    subject_id,
    db_created_date,
    db_updated_date,
    db_deleted_date,
    -- Считаем чистую разницу в минутах (вы можете изменить на секунды или часы)
    EXTRACT(EPOCH FROM (notification_date - target_db_date)) / 60 AS minutes_difference
FROM 
    parsed_notifications
WHERE 
    -- 1. Уведомление строго ПОЗЖЕ, чем дата в БД
    notification_date > target_db_date
    
    -- 2. Разница составляет, например, больше 1 минуты (чтобы отсечь нормальный лаг системы)
    -- Если вам нужен абсолютно любой рассинхрон, поставьте просто > 0
    AND EXTRACT(EPOCH FROM (notification_date - target_db_date)) / 60 > 1.0;
